// ---------- Helpers ----------

export const METRO_BY_ID = new Map(METROS.map((x) => [x.id, x]));

/** Match free text ("Bangalore", "nyc") to a metro by name or alias. */
export function findMetro(text: string): Metro | undefined {
  const q = text.trim().toLowerCase();
  return METROS.find(
    (x) => x.name.toLowerCase() === q || x.aliases.some((a) => a.toLowerCase() === q),
  );
}

/** Nearest metro to a user's coordinates (browser geolocation / IP lookup). */
export function nearestMetro(lat: number, lng: number, maxKm = 150): Metro | undefined {
  const rad = (d: number) => (d * Math.PI) / 180;
  const km = (x: Metro) => {
    const dLat = rad(x.lat - lat), dLng = rad(x.lng - lng);
    const a = Math.sin(dLat / 2) ** 2 + Math.cos(rad(lat)) * Math.cos(rad(x.lat)) * Math.sin(dLng / 2) ** 2;
    return 6371 * 2 * Math.asin(Math.sqrt(a));
  };
  let best: Metro | undefined, bestKm = Infinity;
  for (const x of METROS) {
    const d = km(x);
    if (d < bestKm) { best = x; bestKm = d; }
  }
  return bestKm <= maxKm ? best : undefined; // too far from any metro -> caller falls back to country news
}

/** SerpApi Google News params for a metro's local news. */
export function localNewsParams(x: Metro, englishOnly = true) {
  const names = [x.name, ...x.aliases.filter((a) => a.length > 3)].map((n) => `"${n}"`);
  return {
    engine: "google_news",
    q: names.join(" OR "),              // "Mumbai" OR "Bombay"
    gl: x.countryCode,
    hl: englishOnly ? "en" : x.lang,
  };
}
