enum StockResearchStatus {
  PENDING
  RUNNING
  RERUNNING
  COMPLETED
  FAILED
}

model StockResearch {
  id              String              @id @default(cuid())
  userId          String

  companyName     String?
  symbol          String?
  exchange        String?
  isin            String?
  instrumentKey   String?

  query           String
  status          StockResearchStatus @default(PENDING)

  // Increment whenever the user reruns research.
  runCount        Int                 @default(0)

  determinerPlan  Json?
  report          Json?
  reportMarkdown  String?
  error           String?

  startedAt       DateTime?
  completedAt     DateTime?
  createdAt       DateTime            @default(now())
  updatedAt       DateTime            @updatedAt

  resources       StockResearchResource[]

  @@index([userId, createdAt])
  @@index([status, createdAt])
  @@index([symbol, exchange])
  @@index([isin])
} 

model StockResearchResource {
  id                String                   @id @default(cuid())
  stockResearchId   String

  provider          StockResearchProvider
  type              StockResearchResourceType

  title             String?
  url               String?
  externalId        String?

  // Normalized, readable content for AI synthesis.
  content           String?

  // Original structured API response, article metadata, or AI output.
  rawData           Json?
  metadata          Json?

  // Optional link to Newsly's existing research-source system.
  researchSourceId  String?

  // Useful for deduplication and idempotent retries.
  contentHash       String?

  createdAt         DateTime                 @default(now())
  updatedAt         DateTime                 @updatedAt

  stockResearch     StockResearch            @relation(
    fields: [stockResearchId],
    references: [id],
    onDelete: Cascade
  )

  @@index([stockResearchId, provider])
  @@index([stockResearchId, type])
  @@index([externalId])
  @@index([contentHash])
  @@unique([stockResearchId, provider, contentHash])
}
